    <h2>商圏面積ランキング</h2>

<div id="map"></div>


<script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"></script>

<script src="https://d3js.org/d3.v7.min.js"></script>

<script src="https://unpkg.com/d3-delaunay@6"></script>


<script>

    const map = L.map('map', { doubleClickZoom: false }).setView([35.6581, 139.7017], 15);

    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png').addTo(map);


    let shops = [

        { lat: 35.6585, lng: 139.7010, id: 1 },

        { lat: 35.6600, lng: 139.6980, id: 2 },

        { lat: 35.6570, lng: 139.7040, id: 3 }

    ];


    const svg = d3.select(map.getPanes().overlayPane).append("svg");

    const g = svg.append("g").attr("class", "leaflet-zoom-hide");


    function updateVoronoi() {

        const bounds = map.getBounds();

        const topLeft = map.latLngToLayerPoint(bounds.getNorthWest());

        const bottomRight = map.latLngToLayerPoint(bounds.getSouthEast());

        

        const points = shops.map(d => {

            const p = map.latLngToLayerPoint([d.lat, d.lng]);

            return [p.x, p.y];

        });


        svg.attr("width", bottomRight.x - topLeft.x).attr("height", bottomRight.y - topLeft.y)

           .style("left", topLeft.x + "px").style("top", topLeft.y + "px");

        g.attr("transform", `translate(${-topLeft.x}, ${-topLeft.y})`);


        g.selectAll("*").remove();


        let rankingData = [];


        if (points.length >= 2) {

            const delaunay = d3.Delaunay.from(points);

            const voronoi = delaunay.voronoi([topLeft.x, topLeft.y, bottomRight.x, bottomRight.y]);


            // ボロノイ境界描画 & 面積計算

            const polygons = voronoi.cellPolygons();

            let i = 0;

            for (const poly of polygons) {

                // 面積を計算 (d3.polygonArea)

                const area = Math.abs(d3.polygonArea(poly));

                rankingData.push({ id: i + 1, area: area });


                g.append("path")

                    .attr("d", d3.line()(poly))

                    .attr("stroke", "#2c3e50")

                    .attr("fill", "rgba(231, 76, 60, 0.1)") // 塗りつぶしを追加

                    .attr("stroke-width", 2);

                i++;

            }

        }


        // ランキングの更新

        const list = d3.select("#rank-list");

        list.selectAll("*").remove();

        rankingData.sort((a, b) => b.area - a.area).forEach((d, idx) => {

            list.append("div").attr("class", "shop-item")

                .html(`No.${idx+1} (店舗ID:${d.id})<br>商圏サイズ: <span class="area-val">${Math.round(d.area).toLocaleString()} px²</span>`);

        });


        // 店舗ドットの描画

        points.forEach((p, index) => {

            const circle = g.append("circle")

                .attr("cx", p[0]).attr("cy", p[1]).attr("r", 6)

                .attr("fill", "#e74c3c").attr("stroke", "#fff").attr("stroke-width", 2);


            circle.on("click", (event) => {

                if (event.shiftKey) {

                    event.stopPropagation();

                    shops.splice(index, 1);

                    updateVoronoi();

                }

            });

        });

    }


    map.on("click", (e) => {

        if (!e.originalEvent.shiftKey) {

            shops.push({ lat: e.latlng.lat, lng: e.latlng.lng, id: shops.length + 1 });

            updateVoronoi();

        }

    });


    map.on("viewreset move", updateVoronoi);

    updateVoronoi();

</script>
